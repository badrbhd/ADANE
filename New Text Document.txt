Feature 1: Prayer Times Engine & Adhan Notifications
Calculates times using astronomical algorithms based on coordinates and juristic settings, and schedules foreground/background Adhan alerts.
Dart
// lib/features/prayer_times/prayer_service.dart
import 'package:adhan/adhan.dart';
import 'package:geolocator/geolocator.dart';
import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:timezone/timezone.dart' as tz;

class PrayerService {
  final FlutterLocalNotificationsPlugin _notificationsPlugin = FlutterLocalNotificationsPlugin();

  Future<PrayerTimes> calculatePrayerTimes(Position position) async {
    final coordinates = Coordinates(position.latitude, position.longitude);
    
    // Choose appropriate calculation method (e.g., Muslim World League, Umm al-Qura)
    final params = CalculationMethod.muslim_world_league.getParameters();
    params.madhab = Madhab.shafi; // or Madhab.hanafi for Asr timing

    final date = DateComponents.from(DateTime.now());
    return PrayerTimes(coordinates, date, params);
  }

  Future<void> schedulePrayerNotifications(PrayerTimes times) async {
    final prayers = {
      'Fajr': times.fajr,
      'Dhuhr': times.dhuhr,
      'Asr': times.asr,
      'Maghrib': times.maghrib,
      'Isha': times.isha,
    };

    int id = 0;
    for (var entry in prayers.entries) {
      if (entry.value.isAfter(DateTime.now())) {
        await _notificationsPlugin.zonedSchedule(
          id++,
          'Time for ${entry.key}',
          'حي على الصلاة - حي على الفلاح',
          tz.TZDateTime.from(entry.value, tz.local),
          const NotificationDetails(
            android: AndroidNotificationDetails(
              'adhan_channel',
              'Adhan Notifications',
              importance: Importance.max,
              priority: Priority.high,
              sound: RawResourceAndroidNotificationSound('adhan'), // Bundled audio
              playSound: true,
            ),
            iOS: DarwinNotificationDetails(sound: 'adhan.aiff'),
          ),
          androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
          uiLocalNotificationDateInterpretation:
              UILocalNotificationDateInterpretation.absoluteTime,
        );
      }
    }
  }
}

// lib/features/qibla/qibla_math.dart
import 'dart:math' as math;
import 'package:geolocator/geolocator.dart';

class QiblaCalculator {
  static const double meccaLat = 21.422487;
  static const double meccaLng = 39.826206;

  static double calculateQiblaAngle(Position pos) {
    double latRad = pos.latitude * (math.pi / 180.0);
    double lngRad = pos.longitude * (math.pi / 180.0);
    double meccaLatRad = meccaLat * (math.pi / 180.0);
    double meccaLngRad = meccaLng * (math.pi / 180.0);

    double deltaLng = meccaLngRad - lngRad;

    double y = math.sin(deltaLng);
    double x = math.cos(latRad) * math.tan(meccaLatRad) -
        math.sin(latRad) * math.cos(deltaLng);

    double qibla = math.atan2(y, x) * (180.0 / math.pi);
    return (qibla + 360.0) % 360.0;
  }
}

import 'dart:math' as math;

class QiblaCompassView extends StatefulWidget {
  final Position currentPosition;
  const QiblaCompassView({super.key, required this.currentPosition});

  @override
  State<QiblaCompassView> createState() => _QiblaCompassViewState();
}

class _QiblaCompassViewState extends State<QiblaCompassView> {
  late double _qiblaBearing;

  @override
  void initState() {
    super.initState();
    _qiblaBearing = QiblaCalculator.calculateQiblaAngle(widget.currentPosition);
  }

  @override
  Widget build(BuildContext context) {
    return StreamBuilder<CompassEvent>(
      stream: FlutterCompass.events,
      builder: (context, snapshot) {
        if (!snapshot.hasData || snapshot.data?.heading == null) {
          return const Center(child: CircularProgressIndicator());
        }

        double heading = snapshot.data!.heading!;
        // Needle points to Qibla relative to device heading
        double diff = (_qiblaBearing - heading) * (math.pi / 180.0);

        return Center(
          child: Stack(
            alignment: Alignment.center,
            children: [
              // Compass Dial
              Transform.rotate(
                angle: -heading * (math.pi / 180.0),
                child: Image.asset('assets/images/compass_dial.png', width: 300),
              ),
              // Needle pointing to Makkah
              Transform.rotate(
                angle: diff,
                child: Image.asset('assets/images/qibla_needle.png', width: 260),
              ),
              Positioned(
                bottom: 40,
                child: Text(
                  "${_qiblaBearing.toStringAsFixed(1)}° from North",
                  style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 18),
                ),
              )
            ],
          ),
        );
      },
    );
  }
}

import 'package:flutter/material.dart';

class TasbeehCounterScreen extends StatefulWidget {
  final DhikrItem dhikr;
  const TasbeehCounterScreen({super.key, required this.dhikr});

  @override
  State<TasbeehCounterScreen> createState() => _TasbeehCounterScreenState();
}

class _TasbeehCounterScreenState extends State<TasbeehCounterScreen> {
  void _increment() async {
    if (widget.dhikr.currentCount < widget.dhikr.targetCount) {
      setState(() => widget.dhikr.currentCount++);
      
      if (widget.dhikr.currentCount == widget.dhikr.targetCount) {
        // Heavy vibration on reaching goal
        if (await Vibration.hasVibrator() ?? false) {
          Vibration.vibrate(pattern: [0, 200, 100, 200]);
        }
      } else {
        // Light tap feedback
        Vibration.vibrate(duration: 30);
      }
    }
  }

  void _reset() => setState(() => widget.dhikr.currentCount = 0);

  @override
  Widget build(BuildContext context) {
    double progress = widget.dhikr.currentCount / widget.dhikr.targetCount;

    return Scaffold(
      appBar: AppBar(title: const Text('المسبحة الإلكترونية')),
      body: Padding(
        padding: const EdgeInsets.all(24.0),
        child: Column(
          children: [
            Text(
              widget.dhikr.arabicText,
              textAlign: TextAlign.center,
              style: const TextStyle(fontSize: 22, height: 1.6),
            ),

// lib/features/quran/audio/quran_audio_player.dart
import 'package:just_audio/just_audio.dart';
import 'package:audio_service/audio_service.dart';

class QuranAudioController {
  final AudioPlayer _player = AudioPlayer();

  // Quran.com CDN format or EveryAyah endpoints
  String buildAyahAudioUrl(String reciterCode, int surah, int ayah) {
    // Example format: 001001.mp3 for Surah 1 Ayah 1
    String s = surah.toString().padLeft(3, '0');
    String a = ayah.toString().padLeft(3, '0');
    return "https://everyayah.com/data/$reciterCode/$s$a.mp3";
  }

  Future<void> playAyahRange(List<String> audioUrls) async {
    final playlist = ConcatenatingAudioSource(
      children: audioUrls.map((url) => AudioSource.uri(Uri.parse(url))).toList(),
    );

    await _player.setAudioSource(playlist);
    _player.play();
  }

  void pause() => _player.pause();
  void resume() => _player.play();
  void dispose() => _player.dispose();
}
